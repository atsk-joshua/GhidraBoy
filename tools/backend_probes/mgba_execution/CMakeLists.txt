cmake_minimum_required(VERSION 3.21)
project(GhidraBoyMGBABoundedExecution LANGUAGES C CXX)

if(NOT IS_DIRECTORY "${MGBA_SOURCE}")
    message(FATAL_ERROR "Pass -DMGBA_SOURCE=<pinned patched source>")
endif()
set(LIBMGBA_ONLY ON CACHE BOOL "" FORCE)
set(M_CORE_GB ON CACHE BOOL "" FORCE)
set(M_CORE_GBA OFF CACHE BOOL "" FORCE)
set(ENABLE_DEBUGGERS ON CACHE BOOL "" FORCE)
set(ENABLE_GDB_STUB OFF CACHE BOOL "" FORCE)
set(ENABLE_SCRIPTING OFF CACHE BOOL "" FORCE)
add_subdirectory("${MGBA_SOURCE}" mgba)

set(ADAPTER_EXECUTION_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../../debugger/backends/mgba/native")
add_executable(mgba-execution-probe probe.c "${ADAPTER_EXECUTION_DIR}/execution.c")
target_compile_features(mgba-execution-probe PRIVATE c_std_11)
target_include_directories(mgba-execution-probe PRIVATE
    "${ADAPTER_EXECUTION_DIR}" "${MGBA_SOURCE}/include" "${CMAKE_CURRENT_BINARY_DIR}/mgba/include")
target_compile_definitions(mgba-execution-probe PRIVATE $<TARGET_PROPERTY:mgba,COMPILE_DEFINITIONS>)
find_package(Threads REQUIRED)
target_link_libraries(mgba-execution-probe PRIVATE mgba Threads::Threads)
if(NOT MSVC)
    target_compile_options(mgba-execution-probe PRIVATE -Wall -Wextra -Werror)
endif()
