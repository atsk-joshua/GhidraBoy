package fi.gekkio.ghidraboy;
import ghidra.program.model.address.*;
import ghidra.program.model.pcode.*;
import java.util.*;
public class Before {
 public static void main(String[] ignored) {
  var cs=new GenericAddressSpace("const",64,AddressSpace.TYPE_CONSTANT,0);
  var rs=new GenericAddressSpace("register",64,AddressSpace.TYPE_REGISTER,1);
  var op=new PcodeOp(rs.getAddress(0),0,PcodeOp.INT_RIGHT,new Varnode[]{new Varnode(cs.getAddress(128),1),new Varnode(cs.getAddress(64),8)},new Varnode(rs.getAddress(10),1));
  var result=PcodeConstants.evaluate(op,new HashMap<>(),new HashMap<>());
  System.out.println("Reproduction: INT_RIGHT(0x80:1,64:8) expected=0 actual="+result);
  if (!Long.valueOf(0).equals(result)) throw new AssertionError("confirmed host modulo shift defect");
 }
}
