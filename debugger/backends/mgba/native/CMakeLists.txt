cmake_minimum_required(VERSION 3.21)
project(GhidraBoyMGBA LANGUAGES C CXX)
if(NOT IS_DIRECTORY "${MGBA_SOURCE}")
    message(FATAL_ERROR "Pass -DMGBA_SOURCE=<pinned patched source>")
endif()
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(LIBMGBA_ONLY ON CACHE BOOL "" FORCE)
set(M_CORE_GB ON CACHE BOOL "" FORCE)
set(M_CORE_GBA OFF CACHE BOOL "" FORCE)
set(ENABLE_DEBUGGERS ON CACHE BOOL "" FORCE)
set(ENABLE_GDB_STUB OFF CACHE BOOL "" FORCE)
set(ENABLE_SCRIPTING OFF CACHE BOOL "" FORCE)
set(BUILD_SHARED OFF CACHE BOOL "" FORCE)
set(BUILD_STATIC ON CACHE BOOL "" FORCE)
add_subdirectory("${MGBA_SOURCE}" mgba)
add_library(ghigbc_mgba SHARED adapter.c execution.c)
target_compile_features(ghigbc_mgba PRIVATE c_std_11)
target_include_directories(ghigbc_mgba PRIVATE "${MGBA_SOURCE}/include" "${CMAKE_CURRENT_BINARY_DIR}/mgba/include")
target_compile_definitions(ghigbc_mgba PRIVATE $<TARGET_PROPERTY:mgba,COMPILE_DEFINITIONS>)
target_link_libraries(ghigbc_mgba PRIVATE mgba)
if(NOT MSVC)
    target_compile_options(ghigbc_mgba PRIVATE -Wall -Wextra -Werror)
endif()
